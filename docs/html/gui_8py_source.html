107 def run(self):
108 try:
109 logger.info("Starting background inference worker")
110 # Import standalone inference components
111 from inference_module import InferenceModule
112 from inference_module_standalone import (
113 load_models_from_checkpoint,
114 get_gaussian_quadrature_4point,
115 save_dose,
116 save_nifti_with_manifest,
117 )
118
119 # Quadrature setup
120 quad_energies, quad_weights = get_gaussian_quadrature_4point(self.min_e, self.max_e)
121 logger.info(f"Quad energies: {quad_energies}")
122 self._check_canceled()
123
124 # Load CT data
125 path_lower = self.ct_file.lower()
126 if path_lower.endswith('.nii') or path_lower.endswith('.nii.gz'):
127 nii = nib.load(self.ct_file)
128 ct_array = nii.get_fdata().astype(np.float32)
129 affine = nii.affine
130 elif path_lower.endswith('.npy'):
131 ct_array = np.load(self.ct_file).astype(np.float32)
132 affine = None
133 else:
134 raise ValueError(f"Unsupported CT file format: {self.ct_file}")
135 logger.info(f"CT loaded: shape={ct_array.shape}, dtype={ct_array.dtype}")
136 self._check_canceled()
137
138 # To tensor
139 ct_tensor = torch.from_numpy(ct_array).unsqueeze(0).float()
140 logger.info(f"CT tensor shape: {ct_tensor.shape}")
141 self._check_canceled()
142
143 # Load models
144 device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
145 models_by_energy, scale_factor, clip_min_dict, clip_max_dict, dose_normalization_params = load_models_from_checkpoint(
146 self.model_checkpoint,
147 quad_energies,
148 device=str(device),
149 )
150 logger.info("Models loaded from checkpoint")
151 self._check_canceled()
152
153 # Build inference module
154 infer_mod = InferenceModule(
155 models_by_energy=models_by_energy,
156 energies=quad_energies,
157 energy_weights=quad_weights,
158 device=str(device),
159 scale_factor=scale_factor,
160 energy_min=self.min_e,
161 energy_max=self.max_e,
162 clip_min=clip_min_dict,
163 clip_max=clip_max_dict,
164 dose_normalization_params=dose_normalization_params,
165 )
166 logger.info("InferenceModule constructed")
167 self._check_canceled()
168
169 # Run inference
170 target_cube_size = (64, 64, 64)
171 if len(quad_energies) > 1:
172 dose = infer_mod.run_inference(ct_tensor, target_cube_size=target_cube_size)
173 else:
174 energy = quad_energies[0]
175 dose = infer_mod.run_inference_conditioned_on_energy(
176 ct_tensor,
177 energy_value=energy,
178 target_cube_size=target_cube_size,
179 )
180 logger.info("Inference call finished")
181 self._check_canceled()
182
183 # To numpy
184 if isinstance(dose, torch.Tensor):
185 dose_np = dose.detach().cpu().numpy()
186 else:
187 dose_np = np.array(dose, dtype=np.float32)
188 if dose_np.ndim == 4 and dose_np.shape[0] == 1:
189 dose_np = dose_np[0]
190 if dose_np.ndim == 4 and dose_np.shape[0] == 1:
191 dose_np = dose_np[0]
192 logger.info(f"Dose stats: shape={dose_np.shape}, min={dose_np.min():.6f}, max={dose_np.max():.6f}")
193
194 # Save outputs
195 from pathlib import Path
196 output_path = Path(f"dose_output_standalone_gui_{Path(self.ct_file).stem}.npy")
197 save_dose(output_path, dose_np, affine)
198 logger.info(f"Dose saved to {output_path}")
199 try:
200 save_nifti_with_manifest(dose_np, output_path, root_dir=".")
201 except Exception as e:
202 logger.warning(f"Failed to create NIfTI: {e}")
203
204 # Emit success
205 self.finished.emit(str(output_path))
206 except Exception as e:
207 logger.error("Inference failed", exc_info=True)
208 self.error.emit(str(e))
