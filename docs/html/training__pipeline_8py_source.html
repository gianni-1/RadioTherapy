29class EarlyStopping:
30 """
31 Monitors the validation loss and triggers early stopping if the loss
32 the loss does not improve over a defined number of epochs.
33 """
34 def __init__(self, patience=5):
35 """
36 Args:
37 patience (int): Number of epochs permitted without improvement.
38 """
39 self.patience = patience
40 self.best_val_loss = float('inf')
41 self.epochs_no_improvement = 0
42
43 def update(self, val_loss):
44 """
45 Compares the current validation loss with the best loss to date.
46 Args:
47 val_loss (float): Current loss of validation.
48 Returns:
49 bool: If Early Stopping is triggered true otherwise False.
50 """
51 if val_loss < self.best_val_loss:
52 self.best_val_loss = val_loss
53 self.epochs_no_improvement = 0
54 return False
55 else:
56 self.epochs_no_improvement += 1
57 return self.epochs_no_improvement >= self.patience
58
59 def reset(self):
60 """
61 Resets the early stopping parameters.
62 """
63 self.best_val_loss = float('inf')
64 self.epochs_no_improvement = 0

231 def weighted_dose_loss(self, pred, target, weight_factor=10000.0, dose_threshold=1e-6):
232 """
233 Extreme weighted L1 loss for ultra-sparse dose data.
234
235 Args:
236 pred: Predicted dose values
237 target: Ground truth dose values
238 weight_factor: Extreme multiplier for non-zero dose voxels (10,000x)
239 dose_threshold: Minimum value to consider as "real dose"
240 """
241 # Create extreme weight mask
242 # Weights are 10,000x for dose voxels, nearly zero for background
243 weights = torch.where(target > dose_threshold, 100000.0, 0.01)
244
245 # Weighted absolute difference
246 weighted_diff = torch.abs(pred - target) * weights
247
248 # Calculate stats for logging
249 total_voxels = target.numel()
250 dose_voxels = (target > dose_threshold).sum().item()
251
252 if dose_voxels > 0:
253 dose_loss = weighted_diff[target > dose_threshold].mean()
254 background_loss = weighted_diff[target <= dose_threshold].mean()
255
256 logger.debug(f"EXTREME Weighted loss - {dose_voxels}/{total_voxels} dose voxels "
257 f"({100*dose_voxels/total_voxels:.4f}%), "
258 f"dose_loss: {dose_loss:.6f}, bg_loss: {background_loss:.6f}")
259 else:
260 logger.warning("No dose voxels found - using minimal background loss")
261 return weighted_diff.mean() * 0.001 # Minimal loss when no dose
262
263 return weighted_diff.mean()

296 def extreme_dose_focused_loss(self, pred, target, dose_threshold=1e-6):
297 """
298 Combination of extreme weighted and pure masked loss for ultra-sparse data.
299 Uses 100,000x weights and pure MSE loss for maximum signal.
300 """
301 dose_mask = target > dose_threshold
302 dose_voxels = dose_mask.sum().item()
303
304 if dose_voxels == 0:
305 # No dose - minimal loss to keep training alive
306 return torch.tensor(1e-6, device=target.device, requires_grad=True)
307
308 # Pure dose loss (MSE on dose regions only) - better gradients for sparse data
309 masked_pred = pred[dose_mask]
310 masked_target = target[dose_mask]
311 pure_dose_loss = F.mse_loss(masked_pred, masked_target)
312
313 # EXTREME weighted loss for EXTREME sparsity (1,000,000x weight for dose - MASSIVE INCREASE for 0.005% sparsity)
314 sparsity_percent = 100 * dose_voxels / target.numel()
315
316 # Adaptive weight based on actual sparsity level - MUCH MORE AGGRESSIVE
317 if sparsity_percent < 0.006: # Less than 0.006% sparsity (like we see: 0.005%)
318 dose_weight = 1000000.0 # 1M weight for ultra-extreme cases
319 background_weight = 0.0000001 # Nearly ignore background completely
320 elif sparsity_percent < 0.01: # Less than 0.01% sparsity
321 dose_weight = 500000.0 # 500K weight for extreme cases
322 background_weight = 0.000001 # Nearly ignore background completely
323 else:
324 dose_weight = 100000.0 # Original 100K weight
325 background_weight = 0.00001
326
327 weights = torch.where(dose_mask, dose_weight, background_weight)
328 weighted_loss = (torch.abs(pred - target) * weights).mean()
329
330 # Combine: Emphasize pure dose training even more heavily for extreme sparsity
331 if sparsity_percent < 0.006:
332 combined_loss = 0.95 * pure_dose_loss + 0.05 * weighted_loss # 95% pure dose focus for ultra-extreme
333 elif sparsity_percent < 0.01:
334 combined_loss = 0.9 * pure_dose_loss + 0.1 * weighted_loss # 90% pure dose focus
335 else:
336 combined_loss = 0.8 * pure_dose_loss + 0.2 * weighted_loss
337
338 total_voxels = target.numel()
339 logger.debug(f"EXTREME loss - {dose_voxels}/{total_voxels} dose voxels "
340 f"({sparsity_percent:.5f}%), dose_weight: {dose_weight:.0f}, "
341 f"pure_MSE: {pure_dose_loss:.6f}, weighted_{dose_weight:.0f}: {weighted_loss:.6f}, combined: {combined_loss:.6f}")
342
343 return combined_loss

347 def _compute_dataset_dose_stats(self, loader):
348 """
349 Go once over the whole loader to obtain global mean / std of the
350 ground-truth dose. These statistics are required for a stable
351 normalisation of the target and the network output and will be
352 cached in `self.dose_mean` and `self.dose_std`.
353
354 Args:
355 loader (DataLoader): training (or validation) loader that yields
356 dictionaries with a "target" key.
357 """
358 logger.info("[Dose stats] computing dataset-wide mean / std …")
359 sum_val, sum_sq, num_vox = 0.0, 0.0, 0
360 with torch.no_grad():
361 for b in loader:
362 tgt = b["target"] # shape [B,1,D,H,W]
363 sum_val += tgt.sum().item()
364 sum_sq += (tgt ** 2).sum().item()
365 num_vox += tgt.numel()
366
367 mean = sum_val / num_vox
368 var = (sum_sq / num_vox) - mean ** 2
369 std = math.sqrt(max(var, DOSE_EPS))
370
371 self.dose_mean = mean
372 self.dose_std = std
373 # also capture absolute min / max for min‑max normalisation
374 dataset_min = float('inf')
375 dataset_max = float('-inf')
376 for b in loader:
377 tgt = b["target"]
378 dataset_min = min(dataset_min, tgt.min().item())
379 dataset_max = max(dataset_max, tgt.max().item())
380
381 self.clip_min = dataset_min
382 self.clip_max = dataset_max
383 logger.info(
384 f"[Dose stats] mean={mean:.5f} std={std:.5f} "
385 f"min={dataset_min:.5f} max={dataset_max:.5f}"
386 )

388 @staticmethod
389 def _total_variation_loss(img):
390 """
391 3-D total-variation loss encouraging spatial smoothness.
392
393 Args:
394 img (torch.Tensor): tensor of shape [B, C, D, H, W]
395 """
396 dx = torch.abs(img[..., :-1, :, :] - img[..., 1:, :, :]).mean()
397 dy = torch.abs(img[..., :, :-1, :] - img[..., :, 1:, :]).mean()
398 dz = torch.abs(img[..., :, :, :-1] - img[..., :, :, 1:]).mean()
399 return dx + dy + dz

813 Trains diffusion model for one epoch.

816 train_loader (DataLoader): DataLoader for training data.
817 epoch (int): Current epoch index.

822 def __init__(self, diffusion_model, optimizer_diff, energy_min, energy_max, device):
823 """
824 Args:
825 diffusion_model (torch.nn.Module): Das Diffusionsmodell (z. B. ein UNet).
826 optimizer_diff (torch.optim.Optimizer): Optimierer für das Diffusionsmodell.
827 device (torch.device): CPU oder GPU.

829 self.diffusion_model = diffusion_model
830 self.optimizer_diff = optimizer_diff
831 # Min-Max energy normalization parameters
832 self.energy_min = energy_min
833 self.energy_max = energy_max
834 self.device = device
835 # ---------------------------------------------------------------
836 # Determine the cross‑attention dimension expected by the UNet.
837 #
838 # Many UNet variants have several cross‑attention blocks operating
839 # at different channel widths. Each block owns a projection layer
840 # called “to_k”; its *input* dimension equals the context‑vector
841 # length expected by this block. All blocks that consume the same
842 # context embedding should therefore agree on that dimension –
843 # but some architectures prepend an additional projection layer,
844 # so we might see values like 2, 64, 128, …
845 #
846 # Empirically, the *smallest* of these `in_features` values
847 # corresponds to the original conditioning size (in our case
848 # `(energy, bias) = 2`). We therefore collect **all** `to_k`
849 # modules and pick the minimum `in_features`.
850 # ---------------------------------------------------------------
851 cand_dims = set()
852 for name, mod in self.diffusion_model.named_modules():
853 if isinstance(mod, torch.nn.Linear) and (name.endswith("to_k") or ".to_k" in name):
854 cand_dims.add(mod.in_features)
855
856 if cand_dims:
857 # choose the smallest dimension – typically 2
858 self.cross_attn_dim = min(cand_dims)
859 else:
860 raise RuntimeError(
861 "Could not find any cross-attention projection layer in the diffusion model. "
862 "Ensure that the model is correctly configured for energy conditioning."
863 )
864
865 if self.cross_attn_dim is None:
866 raise RuntimeError("Could not determine cross‑attention dimension from diffusion model.")
867
868 # Note: Context projection disabled since running in unconditional mode
869 # Small MLP that maps (energy, bias) → vector of length `cross_attn_dim`
870 # self.context_proj = torch.nn.Linear(2, self.cross_attn_dim).to(self.device)

874 Args:

876 optimizer_diff (torch.optim.Optimizer): Optimizer for the diffusion model.
877 device (torch.device): CPU or GPU.
878 """
879 # if no inferer or autoencoder provided, skip diffusion training
880 if inferer is None or autoencoder is None:
881 return 0.0
882
883 self.diffusion_model.train()
884 epoch_loss = 0.0
885
886 for step, batch in enumerate(tqdm(train_loader, desc=f"Diffusion Epoch {epoch}")):
887 images = batch["input"].to(self.device)
888
889 # Apply energy conditioning consistently
890 if "energy" in batch:
891 energies = batch["energy"].to(self.device)
892 # Min-Max normalize energy using dataset range with zero-division guard
893 eps = 1e-8

899 conditioned_input = torch.cat([images, energy_tensor], dim=1)

901 # If energy conditioning is expected but not provided, raise an error
902 raise ValueError(
903 "Energy conditioning is expected but 'energy' field is missing in the batch."
904 )
905
906 # Zero gradients for diffusion optimizer
907 self.optimizer_diff.zero_grad(set_to_none=True)
908
909 # Encode conditioned input to latents using the autoencoder
910 with torch.no_grad():
911 # MONAI's AutoencoderKL.encode() returns a tuple (mu, logvar)
912 # For consistent handling, always expect this format
913 mu, logvar = autoencoder.encode(conditioned_input)
914 # Sample from the latent distribution
915 std = torch.exp(0.5 * logvar)
916 eps = torch.randn_like(std)
917 latents = mu + eps * std
918 latents = latents.to(self.device)
919
920 # Crop / pad latents symmetrically so that spatial dims are multiples

925

929 new_W = (W // downsample_factor) * downsample_factor
930

934 w0 = (W - new_W) // 2

940

944 f"but autoencoder produces {latents.shape[1]} channels"
945 )

948 noise = torch.randn_like(latents)

954 noisy_latents = inferer.scheduler.add_noise(latents, noise, timesteps)

956 # ------------------------------------------------------------------
957 # Note: Running in unconditional mode (with_conditioning=False)
958 # Context/conditioning has been disabled in system_manager.py
959 # ------------------------------------------------------------------
960 # energy_feat = normalized_energy.view(B, 1) # [B,1]

967 # if torch.rand(1, device=self.device).item() < 0.1:
968 # context_tensor = None # unconditional pass
969
970 # Predict the noise component and optimise (unconditional)
971 noise_pred = self.diffusion_model(
972 noisy_latents,
973 timesteps=timesteps
974 )
975
976 # ----------------- optimisation step ---------------------------
977 loss = F.mse_loss(noise_pred.float(), noise.float())

979

981 clip_grad_norm_(self.diffusion_model.parameters(), max_norm=1.0)
982
983 self.optimizer_diff.step()

985 if noise.ndim == noise_pred.ndim and noise.shape[2:] != noise_pred.shape[2:]:
986 _, _, Dn, Hn, Wn = noise_pred.shape
987 d0 = (noise.shape[2] - Dn) // 2

989 w0 = (noise.shape[4] - Wn) // 2
990 noise = noise[:, :, d0:d0 + Dn, h0:h0 + Hn, w0:w0 + Wn]
