10class InferenceModule:
11 """
12 The InferenceModule performs the dose inference process using the trained autoencoder
13 and diffusion model. It computes the dose distribution from the uploaded patient CT data.
14
15 The process is as follows:
16 1. Preprocess the input CT scan by resizing or cropping it to a specified cube size.
17 2. Optionally encode the CT scan with the autoencoder to obtain a latent representation.
18 3. Pass the (latent) representation through the diffusion model to generate a voxel-wise dose distribution.
19
20 Additional parameters (e.g. cube size) are taken into account during preprocessing.
21 """

326 def run_inference_over_energies(self, ct_tensor, target_cube_size, energies, energy_weights):
327 """
328 Runs inference on the input CT scan for multiple energy levels and aggregates the results.
329
330 For each energy value provided, this method performs inference (using the conditioned inference
331 method, if available) and then computes a weighted sum of the dose distributions according to the
332 provided energy weights.
333
334 Args:
335 ct_tensor (torch.Tensor): Input CT scan tensor with shape [C, D, H, W].
336 target_cube_size (tuple): Desired spatial dimensions for inference.
337 energies (list of float): A list of energy levels (e.g., [62, 75, 90]).
338 energy_weights (list of float): Corresponding weights for each energy level.
339
340 Returns:
341 torch.Tensor: The aggregated dose distribution computed as:
342 dose_distribution = sum_i (weight_i * N(E_i))
343 """
344 logger.info(f"Running inference over energies: {energies}")
345 dose_list = []
346 # Loop over each energy level.
347 for energy in energies:
348 logger.info(f"Running inference for energy: {energy} keV")
349 # Run inference conditioned on the given energy.
350 dose = self.run_inference_conditioned_on_energy(
351 ct_tensor,
352 energy,
353 target_cube_size
354 )
355 dose_list.append(dose)
356
357 # Stack the dose outputs along a new dimension (energy dimension).
358 dose_stack = torch.stack(dose_list, dim=0)
359
360 # Normalize energy weights so they sum to 1
361 total_w = sum(energy_weights)
362 if not math.isclose(total_w, 1.0, rel_tol=1e-3):
363 energy_weights = [w / total_w for w in energy_weights]
364
365 # Convert energy weights to a tensor for broadcasting.
366 weights_tensor = torch.tensor(energy_weights, dtype=dose_stack.dtype, device=self.device).view(-1, 1, 1, 1, 1)
367
368 # Compute the weighted sum across energy levels.
369 aggregated_dose = torch.sum(dose_stack * weights_tensor, dim=0)
370 logger.info("Aggregated dose distribution computed over all energies")
371 return aggregated_dose
