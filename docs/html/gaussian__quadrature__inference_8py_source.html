37def load_combined_checkpoint(checkpoint_path: str):
38 """Load combined checkpoint with all energy models."""
39 logger = logging.getLogger(__name__)
40 logger.info(f"Loading combined checkpoint from: {checkpoint_path}")
41
42 try:
43 checkpoint = torch.load(checkpoint_path, map_location='cpu')
44 logger.info(f"✓ Checkpoint loaded successfully")
45
46 if 'models_by_energy' in checkpoint:
47 models_by_energy = checkpoint['models_by_energy']
48 logger.info(f"Found {len(models_by_energy)} energy-specific models")
49 for key in models_by_energy.keys():
50 logger.info(f" - {key}")
51
52 if 'dose_normalization_params' in checkpoint:
53 dose_params = checkpoint['dose_normalization_params']
54 logger.info(f"✓ Found dose normalization parameters for {len(dose_params)} energy/resolution combinations")
55 for key, params in dose_params.items():
56 logger.info(f" {key}: clip_min={params.get('clip_min', 0.0):.6f}, clip_max={params.get('clip_max', 1.0):.6f}")
57 else:
58 logger.warning("No dose normalization parameters found in checkpoint")
59 dose_params = {}
60
61 return checkpoint, models_by_energy, dose_params
62
63 except Exception as e:
64 logger.error(f"Failed to load checkpoint: {e}")
65 raise

68def load_ct(path: Path):
69 """Load CT data from .npy or .nii/.nii.gz file."""
70 logger = logging.getLogger(__name__)
71 suffix = path.suffix.lower()
72
73 if suffix in ['.nii', '.gz'] and has_nibabel:
74 nii = nib.load(str(path))
75 arr = nii.get_fdata().astype(np.float32)
76 affine = nii.affine
77 logger.info(f"Loaded NIfTI: shape={arr.shape}")
78 else:
79 arr = np.load(str(path)).astype(np.float32)
80 affine = None
81 logger.info(f"Loaded NPY: shape={arr.shape}")
82
83 logger.info(f"CT range: {arr.min():.2f} to {arr.max():.2f}")
84 return arr, affine

87def extract_energy_models(models_by_energy: dict, target_energies: list):
88 """Extract models for specific energies."""
89 logger = logging.getLogger(__name__)
90 extracted_models = {}
91 available_energies = []
92
93 for key, model_data in models_by_energy.items():
94 try:
95 # Parse energy from key (e.g., "res64x64x64_e46.53")
96 if '_e' in key:
97 energy_str = key.split('_e')[1]
98 energy = float(energy_str)
99 available_energies.append(energy)
100
101 # Check if this energy is in our target list
102 for target_energy in target_energies:
103 if abs(energy - target_energy) < 0.01: # Close enough
104 extracted_models[target_energy] = {
105 'key': key,
106 'energy': energy,
107 'models': model_data
108 }
109 logger.info(f"✓ Found model for energy {target_energy:.4f} keV (key: {key})")
110 break
111 except Exception as e:
112 logger.warning(f"Could not parse energy from key {key}: {e}")
113
114 logger.info(f"Available energies: {sorted(available_energies)}")
115 logger.info(f"Extracted models for {len(extracted_models)} target energies")
116
117 missing_energies = [e for e in target_energies if e not in extracted_models]
118 if missing_energies:
119 logger.error(f"Missing models for energies: {missing_energies}")
120 raise ValueError(f"Models not found for energies: {missing_energies}")
121
122 return extracted_models

125def run_gaussian_quadrature_inference(ct_array, checkpoint_path, quad_energies, quad_weights,
126 device='cpu', cube_size=(64, 64, 64)):
127 """
128 Run 4-point Gaussian Quadrature Inference.
129
130 Args:
131 ct_array: Input CT array
132 checkpoint_path: Path to combined checkpoint
133 quad_energies: List of 4 quadrature energies [x1, x2, x3, x4]
134 quad_weights: List of 4 quadrature weights [w1, w2, w3, w4]
135 device: PyTorch device
136 cube_size: Target cube size
137
138 Returns:
139 integrated_dose: Final integrated dose distribution
140 """
141 logger = logging.getLogger(__name__)
142
143 logger.info("=" * 60)
144 logger.info("GAUSSIAN QUADRATURE INFERENCE")
145 logger.info("=" * 60)
146 logger.info(f"Quadrature energies: {quad_energies}")
147 logger.info(f"Quadrature weights: {quad_weights}")
148 logger.info(f"Device: {device}")
149 logger.info(f"Target cube size: {cube_size}")
150
151 # Load checkpoint
152 checkpoint, models_by_energy, dose_params = load_combined_checkpoint(checkpoint_path)
153
154 # Extract models for quadrature energies
155 extracted_models = extract_energy_models(models_by_energy, quad_energies)
156
157 # Initialize results storage
158 energy_results = {}
159
160 # Run inference for each quadrature energy
161 for i, (energy, weight) in enumerate(zip(quad_energies, quad_weights)):
162 logger.info("-" * 40)
163 logger.info(f"QUADRATURE POINT {i+1}/4: Energy {energy:.4f} keV, Weight {weight:.6f}")
164 logger.info("-" * 40)
165
166 model_info = extracted_models[energy]
167 model_key = model_info['key']
168
169 # Create energy-specific models dict
170 energy_models = {energy: extracted_models[energy]['models']}
171
172 # Get scale factor
173 scale_factor = extracted_models[energy]['models'].get('scale_factor', 1.0)
174 logger.info(f"Using scale factor: {scale_factor}")
175
176 # Create InferenceModule for this energy
177 inference_module = InferenceModule(
178 models_by_energy=energy_models,
179 energies=[energy],
180 energy_weights=[1.0], # Single energy, so weight=1
181 device=device,
182 scale_factor=scale_factor,
183 dose_normalization_params=dose_params
184 )
185
186 # Run inference
187 logger.info(f"Running inference for energy {energy:.4f} keV...")
188 dose_result = inference_module.run_inference_conditioned_on_energy(
189 ct_tensor=torch.from_numpy(ct_array).unsqueeze(0), # Add batch dimension
190 energy_value=energy
191 )
192
193 # Store result
194 if isinstance(dose_result, torch.Tensor):
195 dose_np = dose_result.cpu().numpy()
196 else:
197 dose_np = np.array(dose_result, dtype=np.float32)
198
199 # Remove batch dimension if present
200 if dose_np.ndim == 5 and dose_np.shape[0] == 1:
201 dose_np = dose_np[0, 0] # Remove batch and channel dims
202 elif dose_np.ndim == 4 and dose_np.shape[0] == 1:
203 dose_np = dose_np[0] # Remove batch dim
204
205 energy_results[energy] = {
206 'dose': dose_np,
207 'weight': weight
208 }
209
210 logger.info(f"✓ Energy {energy:.4f}: dose shape={dose_np.shape}, "
211 f"range=[{dose_np.min():.6f}, {dose_np.max():.6f}], weight={weight:.6f}")
212
213 # Gaussian Quadrature Integration
214 logger.info("=" * 40)
215 logger.info("GAUSSIAN QUADRATURE INTEGRATION")
216 logger.info("=" * 40)
217
218 integrated_dose = None
219 total_weight = 0.0
220
221 for energy, weight in zip(quad_energies, quad_weights):
222 dose = energy_results[energy]['dose']
223
224 if integrated_dose is None:
225 integrated_dose = weight * dose
226 else:
227 integrated_dose += weight * dose
228
229 total_weight += weight
230 logger.info(f"Added energy {energy:.4f} with weight {weight:.6f}")
231
232 logger.info(f"Total weight sum: {total_weight:.6f}")
233 logger.info(f"Final integrated dose: shape={integrated_dose.shape}, "
234 f"range=[{integrated_dose.min():.6f}, {integrated_dose.max():.6f}]")
235
236 # Normalize by total weight (should be close to 1 for proper quadrature)
237 if abs(total_weight - 1.0) > 0.001:
238 logger.warning(f"Total weight {total_weight:.6f} != 1.0, normalizing...")
239 integrated_dose /= total_weight
240
241 return integrated_dose, energy_results

258def create_visualization(integrated_dose, energy_results, output_dir):
259 """Create visualization of quadrature integration results."""
260 logger = logging.getLogger(__name__)
261
262 fig, axes = plt.subplots(2, 3, figsize=(18, 12))
263 fig.suptitle('Gaussian Quadrature Dose Integration', fontsize=16)
264
265 # Individual energy results
266 for i, (energy, data) in enumerate(energy_results.items()):
267 if i >= 4: # Only show first 4
268 break
269 row = i // 2
270 col = i % 2
271
272 dose = data['dose']
273 weight = data['weight']
274
275 # Show middle slice
276 mid_slice = dose.shape[2] // 2
277 slice_data = dose[:, :, mid_slice]
278
279 im = axes[row, col].imshow(slice_data, cmap='hot', origin='lower')
280 axes[row, col].set_title(f'Energy {energy:.2f} keV\nWeight: {weight:.4f}')
281 axes[row, col].axis('off')
282 plt.colorbar(im, ax=axes[row, col], shrink=0.8)
283
284 # Integrated result
285 mid_slice = integrated_dose.shape[2] // 2
286 integrated_slice = integrated_dose[:, :, mid_slice]
287
288 im = axes[0, 2].imshow(integrated_slice, cmap='hot', origin='lower')
289 axes[0, 2].set_title('Integrated Dose\n(Gaussian Quadrature)')
290 axes[0, 2].axis('off')
291 plt.colorbar(im, ax=axes[0, 2], shrink=0.8)
292
293 # Statistics
294 axes[1, 2].axis('off')
295 stats_text = f"""Quadrature Integration Stats:
296
297 Final dose range: [{integrated_dose.min():.3f}, {integrated_dose.max():.3f}]
298 Final dose mean: {integrated_dose.mean():.3f}
299 Final dose std: {integrated_dose.std():.3f}
300
301 Individual contributions:"""
302
303 for energy, data in energy_results.items():
304 dose = data['dose']
305 weight = data['weight']
306 contribution = (weight * dose).max()
307 stats_text += f"\n E={energy:.2f}: max_contrib={contribution:.3f}"
308
309 axes[1, 2].text(0.1, 0.9, stats_text, transform=axes[1, 2].transAxes,
310 fontsize=10, verticalalignment='top', fontfamily='monospace')
311
312 plt.tight_layout()
313 viz_path = os.path.join(output_dir, 'gaussian_quadrature_results.png')
314 plt.savefig(viz_path, dpi=300, bbox_inches='tight')
315 plt.close()
316 logger.info(f"✓ Visualization saved: {viz_path}")
