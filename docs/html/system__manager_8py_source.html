32class SystemManager:
33 """
34 The SystemManager class orchestrates the complete training and inference workflow.
35
36 It iterates over each combination of spatial resolution and energy level:
37 - Loads the dataset using the DataLoaderModule.
38 - Splits the data into training and validation sets.
39 - Initializes and trains the autoencoder (and optionally diffusion model)
40 using early stopping based on validation loss.
41 - Sets a training-completion flag for subsequent inference.
42 """
43 def __init__(self, root_dir, transforms, resolutions, energies, energy_min, energy_max, quad_energies, quad_weights, batch_size, device, num_epochs, learning_rate, patience, cube_size, seed=42):
44 """
45 Initializes the SystemManager with configuration parameters.
46
47 Args:
48 root_dir (str): Root directory of the dataset.
49 transforms (monai.transforms.Compose): Preprocessing transforms to apply.
50 resolutions (list of tuple): List of resolutions (e.g., [(64, 64, 64), (32, 32, 32)]).
51 energies (list of int): List of energy levels (e.g., [62, 75, 90]).
52 batch_size (int): Batch size for DataLoader.
53 device (torch.device): Device (CPU or GPU) for computations.
54 seed (int): Random seed for reproducibility.
55 """
56 self.root_dir = root_dir
57 self.transforms = transforms
58 self.resolutions = resolutions
59 self.energies = energies
60 self.energy_min = energy_min
61 self.energy_max = energy_max
62 self.batch_size = batch_size
63 self.device = device
64 self.seed = seed
65 self.num_epochs = num_epochs
66 self.learning_rate = learning_rate
67 self.patience = patience
68 self.training_complete = False
69 self.stop_training = False
70 # placeholders for trained models
71 self.autoencoder = None
72 self.unet = None
73 self.scheduler = None
74 self.cube_size = cube_size
75 self.quad_energies = quad_energies
76 self.quad_weights = quad_weights
77 #keep models per energy for quadrature inference
78 self.models_by_energy = {}
79 # track individual checkpoints in memory
80 self.saved_ckpts = {}

88 ckpt = {
89 "autoencoder": autoencoder.state_dict(),

91 "optimizer_diff": optimizer_diff.state_dict(),
92 "optimizer_g": optimizer_g.state_dict(),
93 "optimizer_d": optimizer_d.state_dict(),
94 "epoch": epoch,
95 "scale_factor": scale_factor,
96 "clip_min": clip_min,
97 "clip_max": clip_max,
98 "smoothing_kernel": smoothing_kernel,
99 "dose_mean": dose_mean,
100 "dose_std": dose_std,
101 "resolution": res,
102 "energy": energy,
103 "learning_rate": self.learning_rate,
104 "batch_size": self.batch_size,
105 "cube_size": self.cube_size,
106 }
107 # build a resolution string for the key
108 if isinstance(res, (tuple, list)):
109 res_str = "x".join(str(r) for r in res)
110 else:
111 res_str = str(res)
112 key = f"res{res_str}_e{energy:.2f}"

146 def run_training(self):
147 """
148 Executes the full training pipeline over all resolution and energy combinations.
149
150 For each combination, the following steps are executed:
151 1. Load and split the dataset filtered by the current energy.
152 2. Create DataLoaders for training and validation.
153 3. Initialize the autoencoder, discriminator, and diffusion models.
154 4. Run the autoencoder training loop with early stopping based on validation loss.
155 5. Run the diffusion model training loop.
156 6. Save the models for the current configuration.
157
158 After all configurations have been processed, a flag is set to indicate that training is complete.
159 """
160 # Print hyperparameters for verification
161 logger.info("Starting CORRECTED training with hyperparameters:")
162 logger.info(f" batch_size={self.batch_size}, num_epochs={self.num_epochs}, learning_rate={self.learning_rate}, patience={self.patience}, cube_size={self.cube_size}")
163 logger.info(f" resolutions={self.resolutions}, energies={self.energies}")
164
165 # CORRECTED TRAINING STRATEGY: Use single resolution (64x64x64) for all energies
166 target_resolution = (64, 64, 64) # Fixed single resolution for stability
167 logger.info(f" CORRECTED: Using SINGLE resolution {target_resolution} for ALL energies (instead of multi-resolution)")
168 logger.info(f" CORRECTED: Energy Loop -> Resolution Loop (proper architecture)")
169
170 # instantiate models - before the loop to avoid re-instantiation
171 autoencoder = AutoencoderKL(
172 spatial_dims=3,
173 in_channels=2,
174 out_channels=1,
175 num_channels=(32, 32, 32),
176 latent_channels=2,
177 num_res_blocks=1,
178 norm_num_groups=8,
179 attention_levels=(False, False, True),
180 ).to(self.device)
181 discriminator = PatchDiscriminator(
182 spatial_dims=3,
183 num_layers_d=3,
184 num_channels=32,
185 in_channels=1,
186 out_channels=1
187 ).to(self.device)
188 unet = DiffusionModelUNet(
189 spatial_dims=3,
190 in_channels=2,
191 out_channels=2,
192 with_conditioning=False, # Disabled cross-attention conditioning
193 num_res_blocks=1,
194 num_channels=(32, 64, 64),
195 attention_levels=(False, True, True),
196 num_head_channels=(0, 64, 64),
197 ).to(self.device)
198
199 # initialize history lists for plotting
200 ae_train_losses = []
201 ae_val_losses = []
202 gen_losses = []
203 disc_losses = []
204 diff_losses = []
205
206 # CORRECTED TRAINING LOOP: Energy as Outer Loop, Single Resolution
207 for energy in self.energies:
208 if self.stop_training:
209 logger.info("Training aborted by user.")
210 return
211 logger.info(f" --- CORRECTED: Training energy={energy} eV at resolution={target_resolution} ---")
212 logger.info(f" CORRECTED: Pipeline uses consistent resolution: {target_resolution}")
213 # Separate transforms for patch-based training and validation
214 patch_transforms = Compose([
215 EnsureTyped(keys=["input", "target"]),
216 Orientationd(keys=["input", "target"], axcodes="RAS"),
217 Resized(keys=["input", "target"], spatial_size=target_resolution, mode=("bilinear", "nearest")),
218 ScaleIntensityRangePercentilesd(keys="input", lower=0, upper=99.5, b_min=0, b_max=1),
219 ToTensord(keys=["input", "target"]),
220 EnsureTyped(keys=["energy"]),
221 ToTensord(keys=["energy"])
222 ])
223 val_transforms = Compose([
224 LoadImaged(keys=["input", "target"], reader=NumpyReader),
225 EnsureChannelFirstd(keys=["input", "target"]),
226 EnsureTyped(keys=["input", "target"]),
227 Orientationd(keys=["input", "target"], axcodes="RAS"),
228 Resized(keys=["input", "target"], spatial_size=target_resolution, mode=("bilinear", "nearest")),
229 ScaleIntensityRangePercentilesd(keys="input", lower=0, upper=99.5, b_min=0, b_max=1),
230 ToTensord(keys=["input", "target"]),
231 EnsureTyped(keys=["energy"]),
232 ToTensord(keys=["energy"])
233 ])
234
235
236 # Use HotspotPatchDataset for patch-based training
237 logger.info("Using HotspotPatchDataset for patch-based training.")
238 train_ds = HotspotPatchDataset(
239 root_dir=self.root_dir,
240 section=None,
241 patch_size=(32, 32, 32),
242 min_hotspot_voxels=100, # REDUCED: Was 1000, now 100 for low-dose data
243 dose_threshold=0.001, # REDUCED: Was 0.5, now 0.001 for low-dose data
244 max_patches=6,
245 random_patches=0,
246 transforms=patch_transforms,
247 energy=energy,
248 max_patches_per_energy=80 if energy <= 40.0 else 60 # Less patches for high energy
249 )
250 # Validation dataset
251 val_ds = HotspotPatchDataset(
252 root_dir=self.root_dir,
253 section=None,
254 patch_size=(32, 32, 32),
255 min_hotspot_voxels=100, # REDUCED: Was 1000, now 100 for low-dose data
256 dose_threshold=0.001, # REDUCED: Was 0.5, now 0.001 for low-dose data
257 max_patches=2,
258 random_patches=0,
259 transforms=patch_transforms,
260 energy=energy,
261 max_patches_per_energy=20 if energy <= 40.0 else 15 # Less patches for high energy validation
262 )
263 # DataLoader as before
264 from data_management import DataLoaderModule
265 data_module = DataLoaderModule(
266 root_dir=self.root_dir,
267 transforms=patch_transforms
268 )
269 train_loader = data_module.create_data_loader(train_ds, self.batch_size, shuffle=True)
270 val_loader = data_module.create_data_loader(val_ds, self.batch_size, shuffle=False)
271 logger.info(f"Found {len(val_ds)} validation patches for energy={energy}, resolution={target_resolution}")
272
273 scheduler = DDPMScheduler(
274 num_train_timesteps=1000,
275 schedule="scaled_linear_beta",
276 beta_start=0.0015,
277 beta_end=0.0195,
278 )
279
280 # ### Scaling factor
281 #
282 # As mentioned in Rombach et al. [1] Section 4.3.2 and D.1, the signal-to-noise ratio (induced by the scale of the latent space) can affect the results obtained with the LDM, if the standard deviation of the latent space distribution drifts too much from that of a Gaussian.
283 # For this reason, it is best practice to use a scaling factor to adapt this standard deviation.
284 #
285 # _Note: In case where the latent space is close to a Gaussian distribution, the scaling factor will be close to one, and the results will not differ from those obtained when it is not used._
286 #
287
288 # +
289 with torch.no_grad():
290 with autocast('cuda', enabled=True):
291 first_batch = first(train_loader)
292 # Build conditioned input for autoencoder with energy channel if available
293 images = first_batch["input"].to(self.device)
294 energies = first_batch.get("energy", None)
295 if energies is not None:
296 energies = energies.to(self.device)
297 normalized_energy = energies.float() / 100.0 # match training normalization
298 B, C, D, H, W = images.shape
299 energy_tensor = normalized_energy.view(B, 1, 1, 1, 1).expand(B, 1, D, H, W)
300 conditioned = torch.cat([images, energy_tensor], dim=1)
301 else:
302 conditioned = images
303 # Encode to latents using conditioned input
304 z = autoencoder.encode_stage_2_inputs(conditioned)
305
306 # **CRITICAL FIX: Handle NaN in scale factor calculation**
307 z_std = torch.std(z)
308 logger.info(f"Latent std: {z_std.item():.6f}")
309
310 if torch.isnan(z_std) or torch.isinf(z_std) or z_std.item() < 1e-8:
311 logger.warning(f"Invalid latent std ({z_std.item()}), using fallback scale_factor=1.0")
312 scale_factor = 1.0
313 else:
314 scale_factor = 1 / z_std.item()
315 # Clamp scale factor to reasonable range
316 scale_factor = max(min(scale_factor, 10.0), 0.1)
317
318 logger.info(f"Scaling factor set to {scale_factor}")
319
320 # **Additional validation**
321 if not np.isfinite(scale_factor):
322 logger.error(f"Scale factor is not finite: {scale_factor}")
323 scale_factor = 1.0
324
325 inferer = LatentDiffusionInferer(scheduler, scale_factor=scale_factor)
326
327 # AGGRESSIVE learning rate for faster convergence (problems identified)
328 # Current signal ratios 0.03-0.08 too low, need stronger learning
329 aggressive_lr = self.learning_rate * 1.0 # Use full original LR (1e-5)
330 logger.info(f"Using AGGRESSIVE learning rate: {aggressive_lr} (original: {self.learning_rate}) for faster convergence")
331
332 opt_g = Adam(autoencoder.parameters(), lr=aggressive_lr, weight_decay=1e-5) # Reduced weight decay
333 opt_d = Adam(discriminator.parameters(), lr=aggressive_lr, weight_decay=1e-5)
334 opt_diff = Adam(unet.parameters(), lr=aggressive_lr, weight_decay=1e-5)
335
336 # trainers and early stopping
337 ae_trainer = AutoencoderTrainer(autoencoder, discriminator, opt_g, opt_d, self.energy_min, self.energy_max, self.device)
338
339 # **CRITICAL FIX: Calculate dose normalization parameters before training**
340 logger.info("Calculating dose normalization parameters from training data...")
341 try:
342 ae_trainer.calculate_dose_normalization_params(train_loader, energy=energy)
343 except Exception as e:
344 logger.error(f"Failed to calculate dose normalization parameters: {e}")
345 logger.error("This usually indicates problems with the training data.")
346 raise
347
348 # Validate that normalization parameters are reasonable
349 if ae_trainer.clip_min is None or ae_trainer.clip_max is None:
350 raise RuntimeError("clip_min or clip_max is None after calculation")
351
352 if ae_trainer.clip_max <= ae_trainer.clip_min:
353 raise RuntimeError(f"Invalid dose range: clip_max ({ae_trainer.clip_max}) <= clip_min ({ae_trainer.clip_min})")
354
355 # Initialize loss functions after normalization parameters are set
356 ae_trainer.setup_loss_functions()
357
358 logger.info(f"Starting autoencoder training for resolution={target_resolution}, energy={energy}")
359 stopper = EarlyStopping(patience=self.patience)
360 # autoencoder training loop
361 for epoch in range(self.num_epochs):
362 if self.stop_training:
363 logger.info(f"Autoencoder training aborted by user at epoch {epoch} for resolution={target_resolution}, energy={energy}")
364 break
365 train_loss, gen_loss, disc_loss = ae_trainer.train_one_epoch(train_loader, epoch)
366 val_loss = ae_trainer.validate(val_loader)
367 # record losses
368 ae_train_losses.append(train_loss)
369 ae_val_losses.append(val_loss)
370 gen_losses.append(gen_loss)
371 disc_losses.append(disc_loss)
372 if stopper.update(val_loss):
373 logger.info(f"Early stopping autoencoder at epoch {epoch+1} for resolution={target_resolution}, energy={energy}")
374 break
375
376 # now diffusion training
377 logger.info(f"Starting diffusion training for resolution={target_resolution}, energy={energy}")
378 diff_trainer = DiffusionTrainer(unet, opt_diff, self.energy_min, self.energy_max, self.device)
379
380 # diffusion (UNet) training loop
381 for epoch in range(self.num_epochs):
382 if self.stop_training:
383 logger.info(f"Diffusion training aborted by user at epoch {epoch} for resolution={target_resolution}, energy={energy}")
384 break
385 diff_loss = diff_trainer.train_one_epoch(train_loader, epoch, inferer, autoencoder)
386 # record diffusion loss
387 diff_losses.append(diff_loss)
388
389 # plot loss curves for this config
390 Visualization.plot_loss_curves(
391 ae_train_losses, ae_val_losses,
392 gen_losses, disc_losses,
393 diff_losses,
394 resolution=target_resolution, energy=energy
395 )
396
397 # **THIS IS THE CRUCIAL ADDITION:**
398 # Save individual model checkpoint to memory after training each configuration
399 self.save_models(
400 autoencoder=autoencoder,
401 unet=unet,
402 optimizer_diff=opt_diff,
403 optimizer_g=opt_g,
404 optimizer_d=opt_d,
405 epoch=epoch, # last epoch
406 res=target_resolution,
407 energy=energy,
408 dose_mean=getattr(ae_trainer, 'dose_mean', None),
409 dose_std=getattr(ae_trainer, 'dose_std', None),
410 scale_factor=scale_factor,
411 clip_min=getattr(ae_trainer, 'clip_min', 0.0), # FIX: Get from trainer
412 clip_max=getattr(ae_trainer, 'clip_max', None), # FIX: Get from trainer
413 smoothing_kernel=0,
414 )
415 logger.info(f"✓ Model checkpoint saved to memory for resolution={target_resolution}, energy={energy}")
416
417 # after all individual trainings, save a single combined checkpoint
418 self.save_combined_checkpoint()
419 self.training_complete = True
420 logger.info("All training finished for all configurations.")

422 def run_inference(self, ct_file_path, model_checkpoint=None):
423 """
424 Load a CT scan and run inference to compute dose distribution.
425 """
426 logger.info("=" * 60)
427 logger.info("STARTING INFERENCE WORKFLOW")
428 logger.info("=" * 60)
429 logger.info(f"CT file path: {ct_file_path}")
430 logger.info(f"Model checkpoint: {model_checkpoint}")
431 logger.info(f"Device: {self.device}")
432
433 # determine model sources and load checkpoint dict if needed
434 ckpt = None
435 if model_checkpoint is None:
436 if not self.models_by_energy:
437 logger.error("No trained models found and no checkpoint provided")
438 raise RuntimeError("No trained models found. Train the models first.")
439 logger.info("Using pre-trained models from training session")
440 elif isinstance(model_checkpoint, str):
441 logger.info(f"Loading checkpoint from file: {model_checkpoint}")
442 import torch as _torch
443 try:
444 ckpt = _torch.load(model_checkpoint, map_location=self.device, weights_only=False)
445 logger.info("✓ Checkpoint loaded successfully")
446 except Exception as e:
447 logger.warning(f"Direct load failed: {e}")
448 logger.info("Attempting to load via file handle...")
449 with open(model_checkpoint, 'rb') as f:
450 ckpt = _torch.load(f, map_location=self.device, weights_only=False)
451 logger.info("✓ Checkpoint loaded via file handle")
452 elif isinstance(model_checkpoint, dict) and 'autoencoder' in model_checkpoint and 'unet' in model_checkpoint:
453 logger.info("Using provided checkpoint dictionary")
454 ckpt = model_checkpoint
455 else:
456 logger.error("Invalid model checkpoint provided")
457 raise ValueError("Invalid model checkpoint. Provide a path or a dict with 'autoencoder' and 'unet' keys.")
458 # if we have a checkpoint dict, rebuild models from state_dict
459 if ckpt is not None:
460 # --- Combined checkpoint support ---
461 models_by_energy_ckpt = ckpt.get("models_by_energy", None)
462 if models_by_energy_ckpt is not None:
463 logger.info("Detected combined checkpoint with per-energy models, loading all energies")
464 self.models_by_energy.clear()
465 for energy_str, model_dict in models_by_energy_ckpt.items():
466 # Extract energy from key format like "res16x16x16_e46.53" or "res16_e46.53"
467 if "_e" in energy_str:
468 energy_val = float(energy_str.split("_e")[1])
469 else:
470 # Fallback: try to convert the whole string (for backward compatibility)
471 energy_val = float(energy_str)
472 logger.warning(f"Using fallback energy extraction for key: {energy_str}")
473 # instantiate autoencoder
474 ae = AutoencoderKL(
475 spatial_dims=3,
476 in_channels=2,
477 out_channels=1,
478 num_channels=(32, 32, 32),
479 latent_channels=2,
480 num_res_blocks=1,
481 norm_num_groups=8,
482 attention_levels=(False, False, True),
483 ).to(self.device)
484 ae.load_state_dict(model_dict["autoencoder"])
485
486 # instantiate unet
487 un = DiffusionModelUNet(
488 spatial_dims=3,
489 in_channels=2,
490 out_channels=2,
491 with_conditioning=False, # Disabled cross-attention conditioning
492 num_res_blocks=1,
493 num_channels=(32, 64, 64),
494 attention_levels=(False, True, True),
495 num_head_channels=(0, 64, 64),
496 ).to(self.device)
497 un.load_state_dict(model_dict["unet"], strict=False)
498 # scheduler
499 sched = DDPMScheduler(
500 num_train_timesteps=1000,
501 schedule="scaled_linear_beta",
502 beta_start=0.0015,
503 beta_end=0.0195,
504 )
505 self.models_by_energy[energy_val] = (ae, un, sched)
506 logger.info("✓ Loaded combined models for all energies")
507
508 # Update quad_energies and quad_weights based on available models
509 available_energies = sorted(self.models_by_energy.keys())
510 logger.info(f"Available energies in checkpoint: {available_energies}")
511
512 # Validate that all required quad_energies are available in the checkpoint
513 missing_energies = []
514 for energy in self.quad_energies:
515 if energy not in available_energies:
516 missing_energies.append(energy)
517
518 if missing_energies:
519 logger.warning(f"Missing energies in checkpoint: {missing_energies}")
520 logger.warning(f"Available energies: {available_energies}")
521 logger.warning(f"Requested energies: {self.quad_energies}")
522 # Filter out missing energies and adjust weights
523 original_energies = self.quad_energies.copy()
524 original_weights = self.quad_weights.copy()
525
526 filtered_energies = []
527 filtered_weights = []
528 for i, energy in enumerate(original_energies):
529 if energy in available_energies:
530 filtered_energies.append(energy)
531 filtered_weights.append(original_weights[i])
532
533 if filtered_energies:
534 # Renormalize weights
535 total_weight = sum(filtered_weights)
536 self.quad_energies = filtered_energies
537 self.quad_weights = [w/total_weight for w in filtered_weights]
538 logger.info(f"Filtered quad_energies: {self.quad_energies}")
539 logger.info(f"Renormalized quad_weights: {self.quad_weights}")
540 else:
541 raise ValueError(f"No quadrature energies available in checkpoint! Available: {available_energies}, Requested: {original_energies}")
542 else:
543 logger.info("✓ All quadrature energies are available in checkpoint")
544 logger.info(f"Using quad_energies: {self.quad_energies}")
545 logger.info(f"Using quad_weights: {self.quad_weights}")
546
547
548 # Load common parameters by averaging over all energy checkpoints
549 if models_by_energy_ckpt:
550 scale_factors = []
551 dose_means = []
552 dose_stds = []
553 clip_mins = []
554 clip_maxs = []
555 smoothing_kernels = []
556
557 for energy_key, model_dict in models_by_energy_ckpt.items():
558 # Only exclude None values, but include 0.0 and other numeric values
559 sf = model_dict.get("scale_factor")
560 if sf is not None:
561 scale_factors.append(sf)
562
563 dm = model_dict.get("dose_mean")
564 if dm is not None:
565 dose_means.append(dm)
566
567 ds = model_dict.get("dose_std")
568 if ds is not None:
569 dose_stds.append(ds)
570
571 cm = model_dict.get("clip_min")
572 if cm is not None:
573 clip_mins.append(cm)
574
575 # clip_max can legitimately be None
576 clip_maxs.append(model_dict.get("clip_max"))
577
578 sk = model_dict.get("smoothing_kernel")
579 if sk is not None:
580 smoothing_kernels.append(sk)
581

788 This is the new approach that trains one model with energy conditioning,

792 1. Load dataset with ALL energies (no filtering by energy)
793 2. Create DataLoaders for training and validation
