105def validate_input_folder(input_folder):
106 """
107 Validate that the input folder structure is correct.
108
109 Args:
110 input_folder (str): Path to the energy folder
111
112 Returns:
113 tuple: (input_cube_path, output_cube_path, dataset_root)
114
115 Raises:
116 ValueError: If folder structure is invalid
117 """
118 input_folder = Path(input_folder)
119
120 if not input_folder.exists():
121 raise ValueError(f"Input folder does not exist: {input_folder}")
122
123 # Check if it's an energy folder with inputcube and outputcube subdirectories
124 input_cube_path = input_folder / "inputcube"
125 output_cube_path = input_folder / "outputcube"
126
127 if input_cube_path.exists() and output_cube_path.exists():
128 # This is an energy folder
129 dataset_root = input_folder.parent
130 logger.info(f"Found energy folder structure: {input_folder}")
131 logger.info(f"Input cubes: {input_cube_path}")
132 logger.info(f"Output cubes: {output_cube_path}")
133 return str(input_cube_path), str(output_cube_path), str(dataset_root)
134 else:
135 # Treat as direct input cube directory
136 logger.info(f"Treating as direct input cube directory: {input_folder}")
137 return str(input_folder), None, str(input_folder.parent)

140def determine_cube_size(input_folder):
141 """
142 Determine cube size from the first available cube file.
143
144 Args:
145 input_folder (str): Path to input cube folder
146
147 Returns:
148 tuple: Cube size dimensions
149 """
150 # Look for .npy, .nii, or .nii.gz files
151 patterns = ["*.npy", "*.nii", "*.nii.gz"]
152 files = []
153
154 for pattern in patterns:
155 files.extend(glob.glob(os.path.join(input_folder, pattern)))
156
157 if not files:
158 raise ValueError(f"No cube files found in {input_folder}")
159
160 sample_file = files[0]
161 logger.info(f"Determining cube size from: {sample_file}")
162
163 try:
164 if sample_file.lower().endswith(".npy"):
165 arr = np.load(sample_file)
166 else:
167 import nibabel as nib
168 arr = np.asarray(nib.load(sample_file).dataobj)
169
170 cube_size = arr.shape
171 logger.info(f"Detected cube size: {cube_size}")
172 return cube_size[0]
173
174 except Exception as e:
175 raise ValueError(f"Failed to determine cube size from {sample_file}: {e}")

178def get_energy_list(dataset_root):
179 """
180 Get list of available energies from dataset root directory.
181
182 Args:
183 dataset_root (str): Path to dataset root containing energy folders
184
185 Returns:
186 list: List of energy values
187 """
188 try:
189 # List all energy subfolder names (skip hidden files)
190 energy_names = [
191 d for d in os.listdir(dataset_root)
192 if os.path.isdir(os.path.join(dataset_root, d)) and not d.startswith('.')
193 ]
194
195 # Parse numeric energy values from folder names and sort numerically
196 energies_list = sorted([float(name.replace("_", ".")) for name in energy_names])
197 logger.info(f"Found energies: {energies_list}")
198 return energies_list
199
200 except Exception as e:
201 logger.warning(f"Could not determine energies from dataset root: {e}")
202 return [0.0] # Default fallback
