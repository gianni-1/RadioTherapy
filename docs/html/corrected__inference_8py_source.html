154 def preprocess_ct(self, ct_array, target_cube_size=(64, 64, 64)):
155 """
156 Preprocess CT data for inference.
157
158 Args:
159 ct_array: CT data as numpy array
160 target_cube_size: Target size for processing
161
162 Returns:
163 Preprocessed CT tensor
164 """
165 # Convert to tensor
166 if isinstance(ct_array, np.ndarray):
167 ct_tensor = torch.from_numpy(ct_array).float()
168 else:
169 ct_tensor = ct_array.float()
170
171 # Add batch and channel dimensions if needed
172 if ct_tensor.dim() == 3:
173 ct_tensor = ct_tensor.unsqueeze(0).unsqueeze(0) # [1, 1, D, H, W]
174 elif ct_tensor.dim() == 4:
175 ct_tensor = ct_tensor.unsqueeze(0) # [1, C, D, H, W]
176
177 # Resize to target size
178 if ct_tensor.shape[2:] != target_cube_size:
179 ct_tensor = F.interpolate(
180 ct_tensor,
181 size=target_cube_size,
182 mode='trilinear',
183 align_corners=False
184 )
185
186 # Normalize CT values (similar to training preprocessing)
187 # Clamp extreme values and normalize to 0-1 range
188 ct_tensor = torch.clamp(ct_tensor, -1000, 3000) # Typical CT range
189 ct_tensor = (ct_tensor + 1000) / 4000 # Normalize to 0-1
190
191 return ct_tensor.to(self.device)

194 """

203 Returns:
204 Dose distribution tensor
205 """
206 if quad_energies is None or quad_weights is None:

249 def run_inference_single_energy(self, ct_tensor, target_energy=None, target_cube_size=(64, 64, 64)):
250 """
251 Run inference on CT data for a single energy.
252
253 Args:
254 ct_tensor: CT data tensor or numpy array
255 target_energy: Target energy for inference (if None, uses first available)
256 target_cube_size: Target cube size for processing
257
258 Returns:
259 Dose distribution tensor
260 """
261 logger.info("-" * 40)
262 logger.info("SINGLE ENERGY INFERENCE STARTED")
263 logger.info("-" * 40)
264
265 try:
266 # Preprocess CT
267 original_shape = ct_tensor.shape if hasattr(ct_tensor, 'shape') else ct_tensor.shape
268 logger.info(f"Original CT shape: {original_shape}")
269
270 input_tensor = self.preprocess_ct(ct_tensor, target_cube_size)
271 logger.info(f"✓ Input preprocessed: {input_tensor.shape}")
272
273 # Choose energy
274 if target_energy is None:
275 target_energy = self.energies[0]
276 logger.info(f"No target energy specified, using default: {target_energy} keV")
277 else:
278 logger.info(f"Using specified energy: {target_energy} keV")
279
280 # Add energy conditioning
281 B, C, D, H, W = input_tensor.shape
282 normalized_energy = target_energy / 100.0 # Normalize like in training
283 energy_tensor = torch.full((B, 1, D, H, W), normalized_energy, device=self.device)
284 conditioned_input = torch.cat([input_tensor, energy_tensor], dim=1)
285
286 logger.info(f"✓ Energy conditioning added:")
287 logger.info(f" - Input shape: {input_tensor.shape}")
288 logger.info(f" - Energy tensor shape: {energy_tensor.shape}")
289 logger.info(f" - Conditioned input shape: {conditioned_input.shape}")
290 logger.info(f" - Raw energy: {target_energy}, normalized: {normalized_energy:.6f}")
291
292 with torch.no_grad():
293 # 1. Encode with autoencoder
294 logger.info("Step 1: Encoding with autoencoder...")
295 encoded = self.autoencoder.encode(conditioned_input)
296 if isinstance(encoded, tuple):
297 latent = encoded[0]
298 else:
299 latent = encoded.latent_dist.sample()
300
301 logger.info(f"Latent shape: {latent.shape}")
302 logger.info(f"Latent range: {latent.min():.4f} to {latent.max():.4f}")
303
304 # 2. Create context for cross-attention
305 energy_weight = 1.0 # Could be adjusted based on energy
306 context = torch.tensor(
307 [[normalized_energy, energy_weight]],
308 dtype=torch.float32,
309 device=self.device
310 ).unsqueeze(1) # [1, 1, 2] - Add sequence dimension
311
312 logger.info(f"Context shape: {context.shape}")
313 logger.info(f"Context values: {context}")
314
315 # 3. Run diffusion sampling
316 logger.info("Running diffusion sampling...")
317
318 # Create noise for sampling
319 noise_shape = latent.shape
320 noise = torch.randn(noise_shape, device=self.device)
321
322 logger.info(f"Noise shape: {noise_shape}")
323 logger.info(f"Using context: {context}")
324
325 # Use the inferer for proper diffusion sampling
326 # The LatentDiffusionInferer handles the full pipeline internally
327 sampled_output = self.inferer.sample(
328 input_noise=noise,
329 autoencoder_model=self.autoencoder,
330 diffusion_model=self.unet,
331 scheduler=self.scheduler,
332 conditioning=context,
333 mode="crossattn"
334 )
335
336 logger.info(f"Sampled output shape: {sampled_output.shape}")
337 logger.info(f"Sampled output range: {sampled_output.min():.4f} to {sampled_output.max():.4f}")
338
339 # The inferer should return the decoded output directly
340 final_output = sampled_output
341
342 # 5. Take only the dose channel (first channel) if multi-channel
343 if final_output.shape[1] > 1:
344 dose_output = final_output[:, :1] # Take first channel
345 logger.info(f"Took first channel for dose, shape: {dose_output.shape}")
346 else:
347 dose_output = final_output
348
349 logger.info(f"Final dose output shape: {dose_output.shape}")
350 logger.info(f"Final dose range: {dose_output.min():.6f} to {dose_output.max():.6f}")
351
352 # 6. Resize back to original size if needed
353 if hasattr(ct_tensor, 'shape'):
354 original_spatial = ct_tensor.shape if len(ct_tensor.shape) == 3 else ct_tensor.shape[-3:]
355 else:
356 original_spatial = original_shape if len(original_shape) == 3 else original_shape[-3:]
357
358 if dose_output.shape[2:] != original_spatial:
359 logger.info(f"Resizing from {dose_output.shape[2:]} to {original_spatial}")
360 dose_output = F.interpolate(
361 dose_output,
362 size=original_spatial,
363 mode='trilinear',
364 align_corners=False
365 )
366
367 # 7. Remove batch dimension
368 final_dose = dose_output[0, 0] # Remove batch and channel dims
369
370 logger.info(f"Final output shape: {final_dose.shape}")
371 logger.info(f"Final output range: {final_dose.min():.6f} to {final_dose.max():.6f}")
372
373 logger.info("✓ SINGLE ENERGY INFERENCE COMPLETED SUCCESSFULLY")
374 logger.info("-" * 40)
375 return final_dose
376
377 except Exception as e:
378 logger.error("SINGLE ENERGY INFERENCE FAILED")
379 logger.error(f"Error: {e}")
380 logger.error("Single energy inference traceback:", exc_info=True)
381 raise e
