70 def display_sample_images(self, images, title="Sample Images", filename=None):
71 """
72 Displays sample images from the given batch. For 3D images, it shows three orthogonal slices.
73
74 Args:
75 images (torch.Tensor or numpy.ndarray): Images to display. Expected shape is either
76 [batch, channels, depth, height, width] for 3D images or [batch, channels, height, width] for 2D.
77 title (str): Title for the plot.
78 filename (str, optional): If provided, the figure is saved to this file.
79 """
80 # Convert to numpy if input is a tensor
81 if hasattr(images, 'cpu'):
82 images = images.cpu().numpy()
83
84 # For simplicity, take the first image from the batch and its first channel
85 sample = images[0, 0]
86
87 # Check dimensions for 3D image
88 if sample.ndim == 3:
89 depth, height, width = sample.shape
90 # Select slices: middle slice along each axis
91 slice_depth = sample[depth // 2, :, :]
92 slice_height = sample[:, height // 2, :]
93 slice_width = sample[:, :, width // 2]
94
95 fig, axs = plt.subplots(1, 3, figsize=(15, 5))
96 axs[0].imshow(slice_depth, cmap='gray')
97 axs[0].set_title("Depth Slice")
98 axs[0].axis("off")
99 axs[1].imshow(slice_height, cmap='gray')
100 axs[1].set_title("Height Slice")
101 axs[1].axis("off")
102 axs[2].imshow(slice_width, cmap='gray')
103 axs[2].set_title("Width Slice")
104 axs[2].axis("off")
105 plt.suptitle(title, fontsize=20)
106 else:
107 # For 2D images, simply show the first image
108 plt.figure(figsize=(6, 6))
109 plt.imshow(sample, cmap='gray')
110 plt.title(title, fontsize=20)
111 plt.axis("off")
112
113 if filename:
114 plt.savefig(filename)
115 else:
116 plt.savefig('sample_images.png')
117 plt.close()

161def load_and_visualize(nifti_path, ct_scan=None, title="Dose distribution - Axial View"):
162 """
163 Loads a Nifti dose distribution file and visualizes its middle axial slice.
164
165 Args:
166 nifti_path: Path to the NIfTI file
167 title: Title for the visualization
168
169 Returns:
170 matplotlib.figure.Figure: A figure object that can be displayed
171 """
172 if nifti_path.lower().endswith(('.nii', '.nii.gz')):
173 img = nib.load(nifti_path)
174 cube = np.asarray(img.dataobj)
175 else:
176 cube = np.load(nifti_path, allow_pickle=True)
177 # Aggregate over energy dimension and ensure 3D volume
178 if cube.ndim == 5 and cube.shape[1] == 1:
179 cube = cube.squeeze(1)
180 if cube.ndim == 4:
181 cube = np.sum(cube, axis=0)
182 cube = np.squeeze(cube)
183 # Call visualization module for interactive slice viewer and volume rendering
184 interactive_slice_viewer(cube, ct_volume=ct_scan)
185 volume_rendering(cube)

201def interactive_slice_viewer(cube, ct_volume=None, title="Interactive Slice Viewer", show=True):
202 """
203 Launches an interactive Matplotlib window with a slider to browse through axial slices of a 3D volume.
204 """
205 if cube.ndim != 3:
206 raise ValueError("Volume must be a 3D array for interactive slice viewing.")
207
208 # Ensure interactive backend is set properly
209 import matplotlib
210 matplotlib.use('QtAgg')
211 import matplotlib.pyplot as plt
212 from matplotlib.widgets import Slider
213
214 # Clear any existing figures to prevent conflicts
215 plt.close('all')
216 plt.ion()
217 plt.switch_backend('QtAgg')
218 # Initial slice
219 z0 = cube.shape[2] // 2
220 fig, ax = plt.subplots(figsize=(8, 7))
221
222 # Adjust layout for sliders
223 if ct_volume is not None:
224 fig.subplots_adjust(bottom=0.25)
225 else:
226 fig.subplots_adjust(bottom=0.15)
227
228 # Initialize images
229 if ct_volume is not None:
230 base_im = ax.imshow(ct_volume[:, :, z0], cmap='gray')
231 overlay_im = ax.imshow(cube[:, :, z0], cmap='jet', alpha=0.5)
232 else:
233 overlay_im = ax.imshow(cube[:, :, z0], cmap='gray')
234
235 ax.set_title(f"{title} - Slice {z0}")
236 ax.axis('off')
237
238 # Create slice slider
239 slider_ax = fig.add_axes([0.15, 0.02, 0.7, 0.03])
240 slider = Slider(slider_ax, 'Slice', 0, cube.shape[2] - 1, valinit=z0, valstep=1)
241
242 # Create alpha slider if CT volume is provided
243 alpha_slider = None
244 if ct_volume is not None:
245 alpha_ax = fig.add_axes([0.15, 0.10, 0.7, 0.03])
246 alpha_slider = Slider(alpha_ax, 'Alpha', 0.0, 1.0, valinit=0.5, valstep=0.01)
247
248 # Keep references to prevent garbage collection
249 slider._fig = fig
250 if alpha_slider is not None:
251 alpha_slider._fig = fig
252
253 # Define callback functions with error handling
254 def update_slice(val):
255 try:
256 z = int(slider.val)
257 if z < 0 or z >= cube.shape[2]:
258 return
259
260 if ct_volume is not None:
261 base_im.set_data(ct_volume[:, :, z])
262 overlay_im.set_data(cube[:, :, z])
263 else:
264 overlay_im.set_data(cube[:, :, z])
265 ax.set_title(f"{title} - Slice {z}")
266 fig.canvas.draw_idle()
267 except Exception as e:
268 print(f"Error in update_slice: {e}")
269
270 def update_alpha(val):
271 try:
272 if ct_volume is not None and alpha_slider is not None:
273 alpha_val = float(alpha_slider.val)
274 if 0.0 <= alpha_val <= 1.0:
275 overlay_im.set_alpha(alpha_val)
276 fig.canvas.draw_idle()
277 except Exception as e:
278 print(f"Error in update_alpha: {e}")
279
280 # Connect callbacks
281 slider.on_changed(update_slice)
282 if alpha_slider is not None:
283 alpha_slider.on_changed(update_alpha)
284
285 # Ensure widgets are properly initialized
286 fig.canvas.draw()
287
288 # Return objects for testing when show=False
289 if not show:
290 if ct_volume is not None:
291 return fig, slider, alpha_slider, overlay_im
292 return fig, slider, overlay_im
293
294 if show:
295 try:
296 plt.show(block=True)
297 except KeyboardInterrupt:
298 plt.close(fig)

301def volume_rendering(cube, opacity=0.2, surface_count=20):
302 """
303 Performs a 3D volume rendering of a normalized 3D array using Plotly.
304 """
305 if cube.ndim != 3:
306 raise ValueError("Volume must be a 3D array for volume rendering.")
307 import numpy as _np
308 import plotly.graph_objects as go
309
310 # normalize
311 cube_norm = (cube - _np.min(cube)) / (_np.max(cube) - _np.min(cube) + 1e-8)
312 # Downsample volume for performance if too large
313 max_points = 200_000 # limit total points to ~200k
314 n_voxels = cube_norm.size
315 if n_voxels > max_points:
316 factor = int(_np.ceil((n_voxels / max_points) ** (1/3)))
317 cube_norm = cube_norm[::factor, ::factor, ::factor]
318 # thresholds
319 p_low, p_high = _np.percentile(cube_norm, [1, 99])
320 # coordinate grid
321 x, y, z = _np.mgrid[0:cube_norm.shape[0],
322 0:cube_norm.shape[1],
323 0:cube_norm.shape[2]]
324 fig = go.Figure(data=go.Volume(
325 x=x.flatten(), y=y.flatten(), z=z.flatten(),
326 value=cube_norm.flatten(),
327 opacity=opacity,
328 surface_count=surface_count,
329 isomin=p_low, isomax=p_high,
330 colorscale='Viridis'
331 ))
332 fig.update_layout(scene=dict(aspectmode='data'))
333 fig.show(renderer="browser")
