298def load_models_from_checkpoint(checkpoint_path, energies, device='cpu'):
299 """
300 Load models from checkpoint and create models_by_energy dict for InferenceModule.
301
302 Args:
303 checkpoint_path (str): Path to model checkpoint
304 energies (list): List of energy values
305 device (str): Device to load models on
306
307 Returns:
308 dict: models_by_energy dictionary
309 """
310 logger = logging.getLogger(__name__)
311 logger.info(f"Loading checkpoint from: {checkpoint_path}")
312
313 # Load checkpoint
314 try:
315 checkpoint = torch.load(checkpoint_path, map_location=device, weights_only=False)
316 logger.info("✓ Checkpoint loaded successfully")
317 except Exception as e:
318 logger.error(f"Failed to load checkpoint: {e}")
319 raise
320
321 models_by_energy_dict = checkpoint.get("models_by_energy")
322 if not models_by_energy_dict:
323 raise RuntimeError("No models_by_energy found in checkpoint")
324
325 logger.info(f"Found {len(models_by_energy_dict)} energy-specific models in checkpoint")
326 logger.info(f"Available energies in checkpoint: {list(models_by_energy_dict.keys())}")
327
328 # Extract dose normalization parameters from checkpoint
329 dose_normalization_params = checkpoint.get("dose_normalization_params", {})
330 if dose_normalization_params:
331 logger.info(f"✓ Found dose normalization parameters for {len(dose_normalization_params)} energy/resolution combinations")
332 for key, params in dose_normalization_params.items():
333 logger.info(f" {key}: clip_min={params.get('clip_min', 'N/A'):.6f}, clip_max={params.get('clip_max', 'N/A'):.6f}")
334 else:
335 logger.warning("No dose_normalization_params found in checkpoint - using fallback")
336
337 # Load energy-specific models - ONLY use the highest resolution (res100) models
338 models_by_energy = {}
339 scale_factor = None
340 clip_min_dict = {}
341 clip_max_dict = {}
342
343 for energy_str, model_dict in models_by_energy_dict.items():
344 energy_val = float(energy_str.split('_e')[1])
345
346 # Skip energies not in our target list
347 if energy_val not in energies:
348 logger.debug(f"Skipping energy {energy_val} keV (not in requested energies)")
349 continue
350
351 # 🔧 UPDATED: Accept res64x64x64 models (new training format) - skip old res25/res50
352 if not (energy_str.startswith('res64x64x64_') or energy_str.startswith('res100_')):
353 logger.debug(f"Skipping {energy_str} - only using highest resolution models (res64x64x64 or res100)")
354 continue
355
356 logger.info(f"Loading models for energy {energy_val} keV (using {energy_str})...")
357
358 # Extract scale_factor from first model (should be same for all)
359 if scale_factor is None:
360 if 'scale_factor' in model_dict:
361 scale_factor = model_dict['scale_factor']
362 logger.info(f"✓ Scale factor extracted: {scale_factor}")
363 else:
364 logger.warning(f"No scale_factor found in model for energy {energy_val}")
365
366 # Extract clip_min and clip_max for this energy
367 if 'clip_min' in model_dict:
368 clip_min_dict[energy_val] = model_dict['clip_min']
369 else:
370 clip_min_dict[energy_val] = 0.0
371 logger.warning(f"No clip_min found for energy {energy_val}, using default 0.0")
372
373 if 'clip_max' in model_dict:
374 clip_max_dict[energy_val] = model_dict['clip_max']
375 else:
376 clip_max_dict[energy_val] = None
377 logger.warning(f"No clip_max found for energy {energy_val}, using default None")
378
379 # Create separate model instances for this energy
380 energy_autoencoder = AutoencoderKL(
381 spatial_dims=3,
382 in_channels=2, # CT + Energy channel
383 out_channels=1,
384 num_channels=(32, 32, 32),
385 latent_channels=2,
386 num_res_blocks=1,
387 norm_num_groups=8,
388 attention_levels=(False, False, True),
389 ).to(device)
390
391 energy_unet = DiffusionModelUNet(
392 spatial_dims=3,
393 in_channels=2,
394 out_channels=2,
395 with_conditioning=False,
396 num_res_blocks=1,
397 num_channels=(32, 64, 64),
398 attention_levels=(False, True, True),
399 num_head_channels=(0, 64, 64),
400 ).to(device)
401
402 energy_scheduler = DDPMScheduler(
403 num_train_timesteps=1000,
404 schedule="scaled_linear_beta",
405 beta_start=0.0015,
406 beta_end=0.0195
407 )
408
409 # Load state dicts for this energy
410 if 'autoencoder' in model_dict:
411 try:
412 energy_autoencoder.load_state_dict(model_dict['autoencoder'])
413 logger.info(f"✓ Autoencoder state loaded for energy {energy_val}")
414 except Exception as e:
415 logger.error(f"Failed to load autoencoder state for energy {energy_val}: {e}")
416 raise
417 else:
418 logger.warning(f"Autoencoder state not found for energy {energy_val}")
419
420 if 'unet' in model_dict:
421 try:
422 # Filter checkpoint to only matching shapes
423 pretrained_dict = model_dict['unet']
424 model_state_dict = energy_unet.state_dict()
425 filtered_dict = {
426 k: v for k, v in pretrained_dict.items()
427 if k in model_state_dict and model_state_dict[k].shape == v.shape
428 }
429 missing_keys = set(model_state_dict.keys()) - set(filtered_dict.keys())
430 unexpected_keys = set(pretrained_dict.keys()) - set(model_state_dict.keys())
431
432 if missing_keys:
433 logger.warning(f"Missing keys for energy {energy_val}: {len(missing_keys)} keys")
434 if unexpected_keys:
435 logger.warning(f"Unexpected keys for energy {energy_val}: {len(unexpected_keys)} keys")
436
437 energy_unet.load_state_dict(filtered_dict, strict=False)
438 logger.info(f"✓ UNet state loaded for energy {energy_val}")
439 except Exception as e:
440 logger.error(f"Failed to load UNet state for energy {energy_val}: {e}")
441 raise
442 else:
443 logger.warning(f"UNet state not found for energy {energy_val}")
444
445 # Set models to eval mode
446 energy_autoencoder.eval()
447 energy_unet.eval()
448
449 # Store models for this energy
450 models_by_energy[energy_val] = (energy_autoencoder, energy_unet, energy_scheduler)
451
452 if not models_by_energy:
453 raise RuntimeError(f"No models loaded for requested energies {energies}")
454
455 if scale_factor is None:
456 logger.warning("No scale_factor found in any model, using default 1.0")
457 scale_factor = 1.0
458
459 logger.info(f"✓ Loaded energy-specific models for {len(models_by_energy)} energies: {list(models_by_energy.keys())}")
460
461 # Return models_by_energy dict, scale_factor, clip_min_dict, clip_max_dict, and dose_normalization_params
462 return models_by_energy, scale_factor, clip_min_dict, clip_max_dict, dose_normalization_params

465def get_gaussian_quadrature_4point(energy_min=3.47, energy_max=46.53):
466 """
467 Get 4-point Gaussian Quadrature nodes and weights for energy integration.
468 Uses the available energies from the checkpoint with optimized weights for Gaussian dose distribution.
469
470 Args:
471 energy_min (float): Minimum energy value (keV) - not used, kept for compatibility
472 energy_max (float): Maximum energy value (keV) - not used, kept for compatibility
473
474 Returns:
475 tuple: (energies, weights) for 4-point Gaussian Quadrature
476 """
477 # Use the available energies from the checkpoint that correspond to 4-point Gaussian Quadrature
478 # Available in checkpoint: 3.47, 11.50, 15.75, 34.25, 46.53
479 # We'll use 4 of these that best match the quadrature pattern
480 energies = [3.47, 15.75, 34.25, 46.53] # Available energies from checkpoint
481
482 # Final optimized weights for perfect Gaussian bell-curve dose distribution
483 # Maximum emphasis on middle energies for sharpest Gaussian peak
484 # Minimal edge weights for smoothest exponential falloff
485 weights = [2.0, 28.0, 28.0, 2.0] # Ultimate Gaussian-like weighting
486
487 # Normalize weights to sum to 1 for dose integration
488 total_weight = sum(weights)
489 normalized_weights = [w / total_weight for w in weights]
490
491 return energies, normalized_weights
