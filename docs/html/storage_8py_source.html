27 def save_model(self, model, filename):
28 """
29 Saves the model's state dictionary to a file.
30
31 Args:
32 model (torch.nn.Module): The model to be saved.
33 filename (str): The filename (relative to model_dir) to save the model.

35 path = os.path.join(self.model_dir, filename)
36 torch.save(model.state_dict(), path)

39 def load_model(self, model, filename, device=torch.device("cpu")):
40 """
41 Loads a saved model's state dictionary from a file and loads it into the given model.
42
43 Args:
44 model (torch.nn.Module): The model instance to load the state dictionary into.
45 filename (str): The filename (relative to model_dir) to load the model from.
46 device (torch.device): The device to map the model to.
47
48 Returns:
49 torch.nn.Module: The model loaded with the saved state dictionary.
50 """
51 path = os.path.join(self.model_dir, filename)
52 if not os.path.exists(path):
53 raise FileNotFoundError(f"Model file {path} not found!")
54 state_dict = torch.load(path, map_location=device)
55 model.load_state_dict(state_dict)
56 print(f"Model loaded from {path}")
57 return model
